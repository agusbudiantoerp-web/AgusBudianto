name: Perbarui data IDX dan terbitkan situs
on:
  push:
    branches: [main]
  schedule:
    - cron: "0 2-10 * * 1-5"   # tiap jam, 09.00-17.00 WIB, Senin-Jumat (waktu GitHub dalam UTC)
  workflow_dispatch:
permissions:
  contents: read
  pages: write
  id-token: write
concurrency:
  group: pages
  cancel-in-progress: true
jobs:
  deploy:
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.dep.outputs.page_url }}
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: Ambil data IDX
        id: gen
        continue-on-error: true
        run: python tools/build_idx.py
      - name: Lewati penerbitan jika data gagal dan ini jadwal otomatis
        if: steps.gen.outcome == 'failure' && github.event_name == 'schedule'
        run: |
          echo "Data gagal; situs lama dipertahankan."
          exit 1
      - uses: actions/configure-pages@v5
      - uses: actions/upload-pages-artifact@v3
        with:
          path: .
      - id: dep
        uses: actions/deploy-pages@v4
